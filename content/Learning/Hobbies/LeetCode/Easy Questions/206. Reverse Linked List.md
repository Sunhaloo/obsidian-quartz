## Iterative Version

```python
def reverseList(self, head: Optional[ListNode]) -> Optional[ListNode]:
	# check if the linked list is empty or not
	if head is None:
		# if the linked list is emtpy ==> return nothing
		return None

	# if the linked list is not empty

	# create our pointers
	current_ptr = head
	previous_ptr = None

	# iterate through the whole linked list while current pointer is not empty
	while current_ptr:
		# hold the current pointer's neighbour
		next_node_ptr = current_ptr.next

		# perform the reverse operations
		current_ptr.next = previous_ptr
		previous_ptr = current_ptr
		current_ptr = next_node_ptr

	# return the previous pointer which is our new head
	return previous_ptr
```

## Recursive Version

```python
def reverseList(self, head: Optional[ListNode]) -> Optional[ListNode]:
	# helper function that is going to be called recursively
	# the `reverse` function is basically an implementation of the iterative one
	def reverse(current_ptr, previous_ptr):
		# check if the current node is `None` ==> base case
		if current_ptr is None:
			return previous_ptr

		# keep track of the current pointer's neighbour
		next_node_ptr = current_ptr.next

		# reverse the node
		current_ptr.next = previous_ptr

		# call the `reverse` function again ==> sub-case
		return reverse(next_node_ptr, current_ptr)

	# finally; call the actual `reverse` function that is going reverse liked list
	return reverse(head, None)
```
