def minimumDeletions(self, nums: List[int]) -> int:
    # initialise the indices
    min_idx = max_idx = 0

    # iterate through the length of the array
    for i in range(len(nums)):
        # check if the current ( actual ) value at `i` is less that at index `min_idx`
        if nums[i] < nums[min_idx]:
            min_idx = i

        # check if the current ( actual ) value at `i` is less that at index `min_idx`
        elif nums[i] > nums[max_idx]:
            max_idx = i

    # check if the ( actual ) value inside of `min_idx` is greater or less than `max_idx`
    if min_idx > max_idx:
        min_idx, max_idx = max_idx, min_idx


    # finally return the minimum number of deletions ( 3 deletion strategies )
    # 1st argument: remove from left --> will also remove min value if max value is after it
    # 2nd argument: remove from right --> will also remove min value if max value if before it
    # 3rd argument: combination of both!
    return min(max_idx + 1, len(nums) - min_idx, (min_idx + 1 + len(nums)) - max_idx)

def minimumDeletions(self, nums: List[int]) -> int:
    # initialise the indices
    min_idx = max_idx = 0

    # iterate through the length of the array
    for index, number in enumerate(nums):
        # check if the current ( actual ) value at `i` is less that at index `min_idx`
        if number < nums[min_idx]:
            min_idx = index

        # check if the current ( actual ) value at `i` is less that at index `min_idx`
        elif number > nums[max_idx]:
            max_idx = index

    # check if the ( actual ) value inside of `min_idx` is greater or less than `max_idx`
    if min_idx > max_idx:
        min_idx, max_idx = max_idx, min_idx

    # finally return the minimum number of deletions ( 3 deletion strategies )
    # 1st argument: remove from left --> will also remove min value if max value is after it
    # 2nd argument: remove from right --> will also remove min value if max value if before it
    # 3rd argument: combination of both!
    return min(max_idx + 1, len(nums) - min_idx, (min_idx + 1 + len(nums)) - max_idx)

# Divide and Conquer Approach Solution

```python
def minimumDeletions(self, nums: List[int]) -> int:
    # get the length of the array
    nums_length = len(nums)

    # check for the length of the array ( cater for 0, 1 or 2 elements in array )
    if nums_length <= 2:
        # remove everything!
        return nums_length

    # function to recursively find the "right" indices for the maximum and minimum
    def find_max_min_indices(left, right):
        # when there is only 1 element left ( during the recursion process )
        if left == right:
            return left, left

        # when there is only 2 element left ( during the recursion process )
        if (right - left) == 1:
            # check which is greater
            if nums[left] > nums[right]:
                # return maximum then minimum
                return left, right

            else:
                # same thing also; return maximum then minimum
                return right, left

        # do the recursive process
        max_idx1, min_idx1 = find_max_min_indices(left, ((left + right) // 2))
        max_idx2, min_idx2 = find_max_min_indices(((left + right) // 2) + 1, right)

        # get the final indices for the maximum and minimum
        final_max_idx = max_idx1 if nums[max_idx1] > nums[max_idx2] else max_idx2
        final_min_idx = min_idx1 if nums[min_idx1] < nums[min_idx2] else min_idx2

        # return the indices to the "main" function
        return final_max_idx, final_min_idx

    # call the function to find the "right" indices
    max_idx, min_idx = find_max_min_indices(0, nums_length - 1)

    # check if the ( actual ) value inside of `min_idx` is greater or less than `max_idx`
    if min_idx > max_idx:
        min_idx, max_idx = max_idx, min_idx
    

    # 1st strategy: remove everything from the left
    # 2nd strategy: remove everything from the right
    # 3rd strategy: remove with a combination of the above
    return min(
        max_idx + 1,
        nums_length - min_idx,
        (min_idx + 1) + (nums_length - max_idx)
    )

```

# Pairwise Approach Solution

> [!NOTE]
> Compared to the above 'Divide and Conquer' approach; this solution results in a *runtime* of about 31ms!
> 
> Its great an all but given that LeetCode's test cases / arrays are quite small. It does **not** really makes sense to write all these code just to get the same answer!
> 
> > This **solidifies** that the fact is; the algorithms that we choose so solve a problem should **specifically** solve *that* problem in that case!

```python
def minimumDeletions(self, nums: List[int]) -> int:
    # get the length of the array
    nums_length = len(nums)

    # check the length of the array ( cater for 0, 1 and 2 elements )
    if nums_length <= 2:
        return nums_length

    # initialise the required indices
    if nums[0] > nums[1]:
        current_max_idx = 0
        current_min_idx = 1

    else:
        current_max_idx = 1
        current_min_idx = 0

    # get the start index
    start = 2

    # check if length of array is even or odd
    if (nums_length % 2) != 0:
        # array length is odd
        last_val = nums[-1]
        last_idx = nums_length - 1

        if last_val > nums[current_max_idx]: current_max_idx = last_idx
        if last_val < nums[current_min_idx]: current_min_idx = last_idx

        end = last_idx

    else:
        # array length is even
        end = nums_length

    # start the iterating through the array
    for i in range(start, end, 2):
        if nums[i] > nums[i + 1]:
            local_max_idx, local_min_idx = i, i + 1

        else:
            local_max_idx, local_min_idx = i + 1, i

        # compare the ( actual ) maximum and minimum values to get correct indices
        if nums[local_max_idx] > nums[current_max_idx]:
            current_max_idx = local_max_idx

        if nums[local_min_idx] < nums[current_min_idx]:
            current_min_idx = local_min_idx

    # check if the ( actual ) value inside of `current_min_idx` is greater / less than `current_max_idx`
    if current_min_idx > current_max_idx:
        current_min_idx, current_max_idx = current_max_idx, current_min_idx

    return min(
        current_max_idx + 1,
        nums_length - current_min_idx,
        (current_min_idx + 1) + (nums_length - current_max_idx),
    )
```
