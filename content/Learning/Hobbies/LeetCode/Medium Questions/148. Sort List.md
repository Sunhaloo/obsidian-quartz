```python
def sortList(self, head: Optional[ListNode]) -> Optional[ListNode]:
	# helper function to find the middle of a linked list
	def get_middle(head):
		# check if the head is empty
		if head is None:
			# simply return the head itself
			return head

		# if the head is not empty ==> initialise our pointer
		slow_ptr = head
		fast_ptr = head.next

		# iterate through the linked list so that `slow_ptr` find the middle node
		while fast_ptr and fast_ptr.next:
			# move the slow pointer by one
			slow_ptr = slow_ptr.next
			# move the fast pointer by 2 ( so that it reaches the end faster )
			fast_ptr = fast_ptr.next.next

		# finally return the middle node
		return slow_ptr

	# helper function to actually merge the two parts of the linked list together
	def merge(left_part, right_part):
		# create dummy node that will act as our pointer
		tail_ptr = dummy = ListNode()

		# while both parts of the linked list has data
		while left_part and right_part:
			# check if the "value" of left part is smaller than right part
			if left_part.val < right_part.val:
				# therefore add that data
				tail_ptr.next = left_part
				left_part = left_part.next

			# check if the "value" of the right part is smaller than left part
			else:
				# therefore add that data to the right part
				tail_ptr.next = right_part
				right_part = right_part.next

			# therefore, change the position of `tail`
			tail_ptr = tail_ptr.next

		# if there are data left inside the left part
		if left_part:
			# hence, add the remaining sorted data form the left
			tail_ptr.next = left_part

		# if there are data left inside the right part
		if right_part:
			# hence, add the remaining sorted dat from the right
			tail_ptr.next = right_part

		# finally --> return the next node of the dummy node
		return dummy.next

	# the actual `merge_sort` function that will be called on linked list
	def merge_sort(head):
		# check if the head is empty
		if (not head) or (not head.next):
			# meaning that there are not data present inside linked list
			return head

		# if there are data inside the linked list ==> get the middle node
		mid_head = get_middle(head)
		# get the 'head' of the right side of the linked list
		right_head = mid_head.next
		# seperate the original linked list into its two components
		mid_head.next = None

		# call the function itself the split the linked list again
		left_part_llist = merge_sort(head)
		right_part_llist = merge_sort(right_head)

		# therefore, when we have individual data ==> list cannot be further splitted
		return merge(left_part_llist, right_part_llist)

	# hence call the `merge_sort` function on the linked list itself and return the sorted linked list
	return merge_sort(head)
```
